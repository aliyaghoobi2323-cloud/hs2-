type wcStream struct {
	id       uint32
	read     uint64        // bytes its reader took
	upd      uint32        // window updates sent
	hit      uint32        // ... of them in cycles where the sender used all of the window
	lag      uint32        // ... in cycles whose reader left over StreamLagTarget/4 unread
	turnMin  time.Duration // least window turnaround (0: no sample)
	turnLast time.Duration // latest window turnaround (0: none)
	win      uint32        // its window now
	braked   bool          // checkLag's brake is on
}
type wcSnd struct { // the edge's view of the exit's socket (the 96-byte stats record)
	fresh        bool
	srtt, minRTT time.Duration
	retrans      uint64 // resent segments since the previous record
	segsOut      uint64 // data segments sent since the previous record
}
type wcIn struct {
	now       time.Time
	minRTT    time.Duration // the link socket's tcpi_min_rtt (0: unknown)
	srtt      time.Duration // the link socket's tcpi_rtt (0: unknown)
	segsIn    uint64        // data segments received
	ooo       uint64        // ... that arrived behind a hole (Rcv_ooopack)
	oooOK     bool
	linkBytes uint64 // bytes the session received
	streams   []wcStream
	parks     uint32          // the session's reader waited for buffer space (borrows taken out)
	brakes    uint32          // checkLag braked a stream
	poolFree  int64           // what the window pool has left (< 0: no pool)
	poolShort uint32          // windows the pool gave less than asked
	ctrlMin   time.Duration   // edge: the control channel's rttMin (0: none)
	pongs     []time.Duration // edge: pong round trips in the interval, oldest first
	snd       wcSnd           // edge: the exit's record
	apProbing bool            // edge: the autopilot is probing the pool size (freeze)
}
type wcCeil struct {
	id   uint32
	ceil uint32 // 0: no ceiling (the behaviour without the core)
}
type wcOut struct {
	state   wcState       // OFF / STARTUP / HOLD / PROBE / DRAIN
	B       uint64        // the published budget (the ramp's)
	ceils   []wcCeil      // one per input stream
	turnRef time.Duration // for Session.SetCoreRef
	qBrake  time.Duration
	growOK  bool
	next    time.Time     // when to call step again (≤ 100 ms while active)
	events  []wcEvent     // filled, drain(why), lossy, pool, capped, route
}
// what the runner feeds wcBuilder.build (maps 1:1 to smux.CoreSample/CoreCounters)
type wcStreamSnap struct {

	braked            bool
}

	apProbing              bool
}

## همهٔ تغییرهای طراحی (و دلیل هر کدام)
1. «فرستندهٔ محدود به پنجره در STARTUP بی‌بی‌آر می‌ماند»: هر اعتبار تازه با ۲٫۸۹ برابر نرخ فرستاده می‌شود. فرض سند (PROBE_BW با cwnd=2BDP) فقط برای فرستندهٔ محدود به cwnd (مثل N4) درست است.
2. «مارپیچ اتلاف خودنگه‌دار» (اتلاف → آزادشدن یک‌جای دادهٔ پشت حفره → UPDهای هم‌زمان → انفجار اعتبار → سرریز تازه؛ بالای حدود ۱٫۵ برابر صف، حتی زیر زانو): مرجع اتلاف وابسته به بودجه (`wcLossRef`)؛ پرش اتلاف در بودجهٔ ثابت با تخلیه تا کف آزموده می‌شود.
3. «رمپ»: سقف‌ها حداکثر max(15ms×R̂, B/8) در هر بیدارباش ۱۰۰ms بالا می‌روند (نزدیک زانو نصف)، پایین آمدن فوری. میان ارزیابی‌ها ورودی‌ها انباشته می‌شوند (`wcMerge`).
4. «F پایدار»: عضو F تا وقتی می‌خواند و خواننده‌اش عقب نیست می‌ماند.
5. «بودجهٔ اثبات‌شده» (`bGood`): HOLD از بزرگ‌ترین بودجه‌ای که ثمر داده بیش از حاشیهٔ h بالاتر نمی‌رود.
6. «q_snd به‌جای q_turn زیر HOL» (لبه، HOLD): رکورد خروجی بی‌صف یعنی چرخش طولانی HOL است.
7. «تازه‌سازی» هر ۷ تا ۹ ثانیه یک دوره زیر HOLD (رویداد DRAIN نمی‌سازد): بی‌آن PROBE_RTT فرستنده و سپس انفجار اعتبار جمع‌شده.
8. کاوش به اندازهٔ زمان صف روی مسیر؛ معیار ثمر با انتظار واقعی.
9. OFF دو دورهٔ پنجرهٔ ثابت (و روی لبه نخستین رکورد خروجی) صبر می‌کند.
10. «برش سریع» میان ارزیابی‌ها: اتلاف روی مسیر پاک (ooo از مرجع پاک گذشت) B را در یک چرخش نصف می‌کند، سقف و Q_cap=1.5·qTarget ثبت می‌شود. مارپیچ فقط اگر پیش از آزادشدن دادهٔ پشت حفره بریده شود می‌ایستد.
11. «تخلیهٔ اتلاف»: پس از DRAIN اتلاف، HOLD تا سه دوره در همان بودجه می‌ماند تا اتلاف تمام شود.
12. «سقف کهنه»: سه دورهٔ پیاپی محدود به پنجره، بی‌صف و بی‌ooo، و ≥۱۵ ثانیه پس از ثبت سقف → سقف و عقب‌نشینی کاوش پاک می‌شود (مسیر تند شده یا رقیب رفته).
13. «آسیب نزن در گام ۱» (OFF): اگر پنجره‌های پایه خودشان در خروجی صف ایستا می‌سازند (q_snd > qTarget+Δq) یا با صف پشت سرشان اتلاف دارند (مرجع اتلاف پایه > ۰٫۳٪ و q_snd ≥ Δq)، هسته شروع نمی‌کند: زیر آن‌ها در گام ۱ چیزی در دست ما نیست. ۱۰۰/۵۰، ۱۰۰/۱۰۰ و ۴×۳۰۰/۱۵۰ را از «بدتر از پایه» بیرون آورد.
14. «STARTUP زیر حفره»: روی مسیر پرتلفات (ooo ≥ ۱۰٪) طولانی‌شدن چرخش صف شمرده نمی‌شود؛ تا رکورد خروجی بعدی صبر می‌کند و اگر صف نشان نداد ادامه می‌دهد. از ناوردایی «اتلاف تصادفی بی‌صف هرگز DRAIN نمی‌سازد» (بذرهای ۳، ۹، ۱۲) آمد.
15. «کاوش ناموفق برمی‌گردد»: کاوش ناموفق یا بی‌نتیجه B را به بودجهٔ شروعش (probeFrom) برمی‌گرداند؛ R̂ که از کاوش بالا رفته بخشی از بودجهٔ بی‌ثمر را نگه نمی‌دارد. کاوشی که F در آن خالی شد تمام می‌شود (پیش‌تر دورهٔ بعد ادامه می‌یافت). از ناوردایی «هر کاوش ناموفق در یک دوره برمی‌گردد».
16. برآورد زمان باقی‌ماندهٔ رمپ نیمه‌شدن گام نزدیک زانو را حساب می‌کند (زمان ارزیابی درست).
17. «منبع کند از F بیرون»: (هشدار بازوی smux) جریانی که در یک دوره هیچ چرخهٔ محدود به پنجره ندارد، چرخشش بیش از دو برابر پایه است، نرخش زیر یک‌چهارم پنجره در یک رفت‌وبرگشت است و (روی لبه) رکورد خروجی صف نشان نمی‌دهد، از F بیرون می‌رود. نسخهٔ اول بدون دو شرط آخر ۳۰۰←۵۰←۳۰۰ را خراب کرد (مسیر کندشده هم چرخش بلند و hit صفر در مدل دارد).
18. رکورد خروجی قدیمی (srtt صفر) نه q_snd می‌دهد نه OFF را منتظر مرجع اتلاف نگه می‌دارد.
19. سازنده: پارک = Δ(Parks − Borrows)؛ PoolShort از شمارندهٔ نشست؛ جریانی که CoreSnapshot کنار گذاشت و برگشت از عکس‌لحظهٔ بعد شمرده می‌شود.
20. در گام ۱ سقف‌های ≤ need به‌صورت ۰ (بی‌سقف، بی‌ترمز) منتشر می‌شوند.
21. دو ثابت بی‌استفاده (wcStartGain، wcStartGainFast) حذف شد؛ پله‌های STARTUP در wcGears = {1.25, 1.5, 2}.

## هدف‌های پذیرش (§۹.۴) در مدل
- برآورده: سود ≥ ۸۰٪ سود N4 روی ۳۰۰/۱۵۰، ۱۰۰۰/۱۰۰، ۱۰۰۰/۱۵۰، ۳۰۰/۱۰۰ و ۱۰۰۰/۵۰۰ (هسته در همه از N4 بیشتر است: ۱۴۵ تا ۲۰۱٪ سود N4)؛ بازارسال ≤ پایه + ۳ در همهٔ خانه‌های جدول پیش‌فرض؛ ≤ پایه + ۰٫۵ جایی که پایه حدود صفر است (به‌جز پلیسر)؛ ۱۰۰/۱۰۰ و ۱۰۰/۱۵۰ کندتر از پایه نیستند؛ تغییر سرعت: بازگشت به ۹۰٪ در ۶٫۴s با بازارسال ۰٫۹۶٪؛ ۴ دانلود روی ۱۰۰۰/۱۰۰ بازارسال برابر پایه؛ صفر DRAIN نادرست زیر اتلاف تصادفی بی‌صف (ناوردایی، ۶۰ بذر).
- برآورده‌نشده: p99 تعاملی ≤ ۱٫۳× پایه در ۳۰۰/۱۵۰ (۲۴۷ در برابر ۱۹۶)، ۱۰۰۰/۱۰۰ (۲۱۱ در برابر ۱۳۱)، ۳۰۰/۱۰۰ (۱۳۶ در برابر ۱۳۳)، ۳۰۰/۱۵۰ q200 و خانه‌های پرتلفات؛ اتلاف ۰٫۱٪: ۳۹ و ۴۰٪ سود N4 (هدف ۶۰٪)؛ ۴ دانلود روی ۳۰۰/۱۵۰: ۲۱٫۱۸٪ در برابر ۲۰٫۵۸٪ پایه (هدف ≤ پایه)؛ پلیسر کندتر از پایه؛ در ماتریس کامل ۷ خانه از حد نقشهٔ راه (پایه + ۳) می‌گذرند؛ «صفر DRAIN از HOLD روی مسیر ثابت پاک» در ۳۰۰/۱۵۰، ۱۰۰۰/۱۰۰، ۳۰۰/۱۵۰ q200 و ۴×۱۰۰۰/۱۰۰ (یک یا دو).
- در مدل سنجیدنی نیست: پردازنده بر مگابایت، ماتریس توقف و بریدن، آپلود (سمت خروجی)، سازگاری نسخه‌ها.

## آزمون‌ها و زمان
- `go test ./engine/ -run Wcore` (پیش‌فرض): حدود ۱٫۷ ثانیه (جدول ۱۸ خانه × ۳ اجرا حدود ۱٫۵s، ناوردایی ۱۲ بذر ۰٫۱۲s، آزمون‌های واحد میلی‌ثانیه). هر اجرای ۳۰ ثانیه‌ای مدل چند ده میلی‌ثانیه (زیر ۱s سنجیده می‌شود).
- `HS2_WCORE_MODEL_FULL=1`: ۸۵ خانه و ۶۰ بذر، حدود ۷ ثانیه.
- `-race` روی آزمون‌های هسته: حدود ۳۰ ثانیه، سبز.
- `HS2_WCORE_MODEL_TRACE=rate/rtt/queue[/loss[/mode[/streams]]]` ردگیری یک خانه.
- جدول پیش‌فرض دروازه است: هر کمبودی بیرون از `wmKnown` شکست است؛ خانهٔ فهرست‌شده‌ای که همه را برآورد گزارش می‌شود. ماتریس کامل گزارش است و فقط حد بازارسال نقشهٔ راه را با `wmKnownRoadmap` می‌سنجد.
- `go vet ./engine/` تمیز. `go test ./engine/` کامل یک بار: فقط `TestWedgeGuardTricklingReaderDoesNotHideStall` شکست خورد؛ بدون فایل‌های هسته هم دو بار از سه بار زیر بار دستگاه شکست خورد (ناپایدار و قدیمی، ربطی به این کار ندارد).

## مسائل باز برای ادغام
1. اجراکننده: یکی برای هر لینک؛ `step` در `out.next` (فعال: هر ۱۰۰ms، OFF: ۴۰۰ms)، در هر بیدارباش یک TCP_INFO (min_rtt، rtt، segs_in، rcv_ooopack)، `CoreSnapshot` و `WindowPool.Free`، رکورد خروجی؛ `wcLinkSnap` از این‌ها ساخته و به `wcBuilder.build` داده می‌شود.
2. سقف‌ها با شناسه نگاشت شوند؛ جریانی که CoreSnapshot کنار گذاشت `SetCoreCeil(0)` بگیرد (مدل همین را می‌کند). `SetCoreRef(turnRef, qBrake, growOK)` در هر گام.
3. تعریف hit در بازو: مدل «فاصله در گیرنده» را جانشین کرده (بایت‌های L_prev−frame دست‌کم ۲ms پیش از نخستین بایت فراتر رسیده‌اند)؛ زیر صف ایستا این نشانه کم‌شمار است (همین قاعدهٔ منبع کند را خراب کرد). تعریف واقعی «Hits» بازو باید با مدل مقایسه شود؛ اگر پس‌ماندهٔ گیرنده را می‌سنجد، باید به نشانهٔ فاصله برگردد.
4. `wcStep` آینهٔ `smux.StepOf` است و مدل `modelStepOf`/`modelFitCore` را خط‌به‌خط از third_party/smux/core.go دارد؛ در ادغام به خود آن‌ها وصل شوند (آزمون `TestWcoreStepOf` برابری را می‌سنجد).
5. q_snd فقط از رکورد تازه (هر حدود ۲ ثانیه) می‌آید؛ قاعده‌های 6، 13، 14 و 17 به آن تکیه دارند. سمت خروجی (آپلود) q_snd ندارد و در مدل سنجیده نشده است (مدل فقط دانلود لبه است).
6. گام ۲ در جدول آزموده نشده (همهٔ خانه‌ها گام ۱)؛ برای صف‌های کوچک روی مسیر بلند همان درمان لازم است.
7. sweepClaims پنجرهٔ بی‌خواندن را پس از ۵ ثانیه به need برمی‌گرداند: جریان بی‌خواندن از F بیرون می‌رود، پس با هسته سازگار است؛ با پنجرهٔ ۱۶MiB فقط حدود نیمی از UPDها نمونهٔ چرخش می‌دهند و دوره‌های بی‌نمونه صف را داوری نمی‌کنند.
8. همهٔ ثابت‌ها متغیر بسته‌اند (برای واسنجی).
